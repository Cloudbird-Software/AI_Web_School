else
  # 记录当前行数（与main分支对比应不减少）
  if git rev-parse --verify main >/dev/null 2>&1; then
    MAIN_LINES=$(git show main:"$FROZEN_FILE" 2>/dev/null | wc -l || echo "0")
    CURRENT_LINES=$(wc -l < "$FROZEN_FILE")
    if [ "$CURRENT_LINES" -lt "$MAIN_LINES" ]; then
      echo "❌ FROZEN.txt行数减少（$MAIN_LINES → $CURRENT_LINES），冻结契约被移除"
      exit 1
    fi
    echo "✅ FROZEN.txt行数未减少（$CURRENT_LINES lines）"
  else
    echo "⚠️  无main分支对比，仅检查文件存在"
  fi
fi

### 12. W1最小链路烟测（内容写入→事件记录）
**验证脚本**（tests/golden-path/test_w1_min_link.py）：
```python
"""W1最小链路：创建门证书→发布item_version(draft→published)→写一条response_event"""
import pytest
import ulid

@pytest.mark.asyncio
async def test_w1_min_content_publish_and_event(async_session):
    """W1端到端最小组合链路验证"""
    from src.core.gate.models import GateCertificate
    from src.core.content.writer import publish_item_version
    from src.core.events.writer import record_event
    from src.core.models.item import Item
    from src.core.models.item_version import ItemVersion
    from sqlalchemy import select, func

    # Step 1: 创建门证书
    cert_id = str(ulid.new())
    cert = GateCertificate(
        id=cert_id, artifact_ref="w1-test", cert_type="smoke",
        policy_version="1.0", issued_by="w1-exit"
    )
    async_session.add(cert)
    await async_session.commit()

    # Step 2: 发布一个published版本（持门证书）
    vdata = {
        "pack_id": "platform",
        "tier": "C",
        "status": "published",
        "objective": {"kp_set": [{"dimension": "kp", "code": "smoke.test"}], "kp_set_mode": "single", "cognitive_level": "remember", "gradeband": "L", "graph_release": "2026.1"},
        "interaction_ref": {"interaction_id": "single_choice", "interaction_params": {}},
        "content": {"blocks": [{"type": "text", "value": "W1出口烟测题"}]},
        "scoring_ref": {"scorer_id": "exact_match", "scorer_params": {"correct_answer": "A"}},
        "error_bindings": [],
        "lineage": {"tier": "C", "pipeline": {"id": "w1-exit", "version": "1.0"}, "signed_by": "smoke", "signed_at": "2026-07-26T00:00:00Z"},
    }
    pub_result = await publish_item_version(
        item_id=None, version_data=vdata,
        gate_certificate_id=cert_id, db=async_session
    )
    assert pub_result["item_id"] is not None
    assert pub_result["item_version_id"] is not None

    # 验证current_version_id已更新
    item = await async_session.get(Item, pub_result["item_id"])
    assert item.current_version_id == pub_result["item_version_id"]

    # Step 3: 写入一条作答事件
    import uuid
    from datetime import datetime, timezone
    event_id = uuid.uuid4()
    eid = await record_event(
        event_id=event_id,
        student_alias_id=uuid.uuid4(),
        item_version_id=pub_result["item_version_id"],
        scene="practice",
        raw_payload={"answer": "A"},
        scoring_trace={"scorer_id": "exact_match", "scorer_version": "1.0", "confidence": {"scoring": 1.0}},
        error_inferences=[],
        created_at=datetime.now(timezone.utc),
        db=async_session
    )
    assert eid is not None

    # Step 4: 验证append-only——尝试UPDATE应被触发器拒绝
    from sqlalchemy import text
    with pytest.raises(Exception):
        await async_session.execute(
            text("UPDATE response_event SET raw_payload = '{}' WHERE event_id = :eid"),
            {"eid": str(event_id)}
        )
    await async_session.rollback()
```

**执行命令**：
```bash
python -m pytest tests/golden-path/test_w1_min_link.py -v
```

**验收标准**：链路全通，UPDATE被触发器阻断。
