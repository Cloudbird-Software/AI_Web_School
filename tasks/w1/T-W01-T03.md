## 验证策略
本验证卡覆盖W1波次的两个主线核心任务：SQLAlchemy ORM实体模型与Pydantic schema（T-W1-003）、内容版本写入服务（T-W1-007）。这两个任务构成数据飞轮的写入核心，含内容寻址与门强制两大关键逻辑。

验证层次：
1. **ORM文件结构验证**：九实体一一对应，每个含Base映射+Pydantic模型
2. **ORM-DDL一致性**：__tablename__/列名/类型与迁移一致
3. **JSONB子模型验证**：六大块字段对应Pydantic子模型，类型校验有效
4. **内容寻址纯函数**：同输入同输出，签名与契约逐字对齐
5. **JSONB序列化往返**：obj→JSON→obj一致
6. **写入服务正常路径**：创建item+version、current_version_id更新正确
7. **门强制规则**：published无证书→失败；draft/quarantined→成功
8. **lineage写入正确**：必填字段齐全，gate_certificate_id在列字段内
9. **版本累积验证**：同一item多次发布产生多个version，旧版不删
10. **素材/语料写入骨架**：publish_material/publish_corpus可用

def test_content_addressing_different_input_different_id():
    base_args = (
        "sha256:template_v1",

        "sha256:engine-v1",
        [],
        "zh-CN"

async def test_jsonb_columns_roundtrip(async_session):

    from src.core.models.item_version import ItemVersion, Objective, InteractionRef, \

@pytest.mark.asyncio
async def test_multiple_publishes_version_accumulation(async_session):
    """同一item两次发布→两个version均存在、current指向最新、旧版不删除"""
    from src.core.content.writer import publish_item_version
    from src.core.models.item import Item
    from src.core.models.item_version import ItemVersion
    from src.core.gate.models import GateCertificate
    import ulid

    # 创建证书
    cert_id1 = str(ulid.new())
    cert_id2 = str(ulid.new())
    for cid in [cert_id1, cert_id2]:
        async_session.add(GateCertificate(id=cid, artifact_ref="t", cert_type="r", policy_version="1", issued_by="v"))
    await async_session.commit()

    vdata_base = {
        "pack_id": "subject-math",
        "tier": "C",
        "objective": {"kp_set": [{"dimension": "kp", "code": "math.test"}], "kp_set_mode": "single", "cognitive_level": "remember", "gradeband": "L", "graph_release": "2026.1"},
        "interaction_ref": {"interaction_id": "single_choice", "interaction_params": {}},
        "content": {"blocks": []},
        "scoring_ref": {"scorer_id": "exact_match", "scorer_params": {}},
        "error_bindings": [],
        "lineage": {"tier": "C", "pipeline": {"id": "test", "version": "1.0"}, "signed_by": "test", "signed_at": "2026-01-01T00:00:00Z"},
    }

    # 第一次发布
    vdata1 = dict(vdata_base, status="published")
    r1 = await publish_item_version(item_id=None, version_data=vdata1, gate_certificate_id=cert_id1, db=async_session)
    item_id = r1["item_id"]
    v1_id = r1["item_version_id"]

    # 第二次发布（修改content）
    vdata2 = dict(vdata_base, status="published", content={"blocks": [{"type": "text", "value": "v2"}]})
    r2 = await publish_item_version(item_id=item_id, version_data=vdata2, gate_certificate_id=cert_id2, db=async_session)
    v2_id = r2["item_version_id"]

    # 验证：两个version都存在
    v1 = await async_session.get(ItemVersion, v1_id)
    v2 = await async_session.get(ItemVersion, v2_id)
    assert v1 is not None, "旧版本v1不应被删除"
    assert v2 is not None
    assert v1_id != v2_id

    # current_version_id指向最新published
    item = await async_session.get(Item, item_id)
    assert item.current_version_id == v2_id
